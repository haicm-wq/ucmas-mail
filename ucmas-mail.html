📋 SQL Scripts — Chạy trong Supabase SQL Editor
Copy từng block hoặc copy toàn bộ → vào Supabase Dashboard → SQL Editor → New query → Paste → Run (F5)
Chạy theo thứ tự: 1. Types → 2. Tables → 3. Indexes → 4. RLS → 5. Seed data
SQL — Create all tables
-- ═══ UCMAS MAIL — Complete Database Schema ═══
-- Run this in Supabase SQL Editor

-- 1. Custom ENUM types
CREATE TYPE contact_status AS ENUM ('active', 'unsubscribed', 'bounced');
CREATE TYPE campaign_status AS ENUM ('draft', 'sending', 'completed', 'failed');
CREATE TYPE send_status    AS ENUM ('sent', 'failed', 'skipped');

-- 2. levels table (self-referencing for sub-levels)
CREATE TABLE levels (
  id          uuid         DEFAULT gen_random_uuid() PRIMARY KEY,
  name        text         NOT NULL UNIQUE,
  color       text         NOT NULL DEFAULT '#5ba8ff',
  parent_id   uuid         REFERENCES levels(id) ON DELETE SET NULL,
  description text,
  sort_order  int4         DEFAULT 0,
  created_at  timestamptz  DEFAULT now()
);

-- 3. contacts table
CREATE TABLE contacts (
  id          uuid           DEFAULT gen_random_uuid() PRIMARY KEY,
  name        text           NOT NULL,
  email       text           NOT NULL UNIQUE,
  level_id    uuid           REFERENCES levels(id) ON DELETE SET NULL,
  company     text,
  phone       text,
  tags        text[]         DEFAULT '{}',
  status      contact_status  DEFAULT 'active',
  last_sent_at timestamptz,
  created_at  timestamptz    DEFAULT now()
);

-- 4. templates table
CREATE TABLE templates (
  id          uuid         DEFAULT gen_random_uuid() PRIMARY KEY,
  name        text         NOT NULL,
  icon        text         DEFAULT '📄',
  description text,
  body        text         NOT NULL,
  tags        text[]       DEFAULT '{}',
  created_at  timestamptz  DEFAULT now(),
  updated_at  timestamptz  DEFAULT now()
);

-- 5. campaigns table
CREATE TABLE campaigns (
  id            uuid            DEFAULT gen_random_uuid() PRIMARY KEY,
  name          text            NOT NULL,
  from_name     text            NOT NULL,
  from_email    text            NOT NULL,
  subject       text            NOT NULL,
  body_text     text            NOT NULL,
  target_levels text[]          DEFAULT '{}',
  status        campaign_status  DEFAULT 'draft',
  sent_count    int4            DEFAULT 0,
  failed_count  int4            DEFAULT 0,
  created_at    timestamptz     DEFAULT now(),
  sent_at       timestamptz
);

-- 6. send_logs table
CREATE TABLE send_logs (
  id           uuid         DEFAULT gen_random_uuid() PRIMARY KEY,
  campaign_id  uuid         NOT NULL REFERENCES campaigns(id) ON DELETE CASCADE,
  contact_id   uuid         REFERENCES contacts(id) ON DELETE SET NULL,
  email        text         NOT NULL,
  level        text,
  status       send_status  NOT NULL,
  resend_id    text,
  error_msg    text,
  sent_at      timestamptz  DEFAULT now()
);
SQL — Indexes (performance)
-- Indexes giúp query nhanh hơn khi filter theo level, email, campaign

CREATE INDEX idx_contacts_level    ON contacts(level_id);
CREATE INDEX idx_contacts_email    ON contacts(email);
CREATE INDEX idx_contacts_status   ON contacts(status);
CREATE INDEX idx_levels_parent     ON levels(parent_id);
CREATE INDEX idx_send_logs_camp    ON send_logs(campaign_id);
CREATE INDEX idx_send_logs_contact ON send_logs(contact_id);
CREATE INDEX idx_send_logs_status  ON send_logs(status);
CREATE INDEX idx_campaigns_status  ON campaigns(status);

-- Auto-update updated_at cho templates
CREATE OR REPLACE FUNCTION update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = now();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER templates_updated_at
  BEFORE UPDATE ON templates
  FOR EACH ROW EXECUTE FUNCTION update_updated_at();

-- Bảng lưu config app (Resend key, FROM_EMAIL, service key)
CREATE TABLE app_config (
  key        text         PRIMARY KEY,
  value      text         NOT NULL,
  updated_at timestamptz  DEFAULT now()
);
ALTER TABLE app_config ENABLE ROW LEVEL SECURITY;
CREATE POLICY "auth_config" ON app_config FOR ALL TO authenticated USING (true);

-- View: contacts với level name (JOIN sẵn)
CREATE VIEW contacts_with_level AS
  SELECT c.*, l.name AS level_name, l.color AS level_color, l.parent_id
  FROM contacts c
  LEFT JOIN levels l ON c.level_id = l.id;
SQL — Row Level Security
-- Bật RLS (Row Level Security) cho mọi bảng
-- Quan trọng: không bật RLS → data public với anon key!

ALTER TABLE levels     ENABLE ROW LEVEL SECURITY;
ALTER TABLE contacts   ENABLE ROW LEVEL SECURITY;
ALTER TABLE templates  ENABLE ROW LEVEL SECURITY;
ALTER TABLE campaigns  ENABLE ROW LEVEL SECURITY;
ALTER TABLE send_logs  ENABLE ROW LEVEL SECURITY;

-- Policy: Chỉ authenticated users được đọc/ghi
-- (Dùng service_role key từ backend để bypass RLS)

CREATE POLICY "auth_read_levels"     ON levels     FOR SELECT TO authenticated USING (true);
CREATE POLICY "auth_write_levels"    ON levels     FOR ALL    TO authenticated USING (true);
CREATE POLICY "auth_read_contacts"   ON contacts   FOR SELECT TO authenticated USING (true);
CREATE POLICY "auth_write_contacts"  ON contacts   FOR ALL    TO authenticated USING (true);
CREATE POLICY "auth_read_templates"  ON templates  FOR SELECT TO authenticated USING (true);
CREATE POLICY "auth_write_templates" ON templates  FOR ALL    TO authenticated USING (true);
CREATE POLICY "auth_read_campaigns"  ON campaigns  FOR SELECT TO authenticated USING (true);
CREATE POLICY "auth_write_campaigns" ON campaigns  FOR ALL    TO authenticated USING (true);
CREATE POLICY "auth_read_logs"       ON send_logs  FOR SELECT TO authenticated USING (true);
CREATE POLICY "auth_write_logs"      ON send_logs  FOR ALL    TO authenticated USING (true);

const supabaseUrl  = process.env.NEXT_PUBLIC_SUPABASE_URL

JavaScript — Contacts CRUD
// Lấy tất cả contacts kèm level info
export async function getContacts(levelId = null) {
  let query = supabase
    .from('contacts')
    .select('*, levels(name, color, parent_id)')
    .eq('status', 'active')
    .order('created_at', { ascending: false })

  if (levelId) query = query.eq('level_id', levelId)

  const { data, error } = await query
  if (error) throw error
  return data
}

// Upsert contacts từ CSV (bulk import)
export async function upsertContacts(contacts) {
  const { data, error } = await supabase
    .from('contacts')
    .upsert(contacts, { onConflict: 'email' })
    .select()
  if (error) throw error
  return data
}

// Cập nhật level cho 1 contact
export async function updateContactLevel(contactId, levelId) {
  const { error } = await supabase
    .from('contacts')
    .update({ level_id: levelId })
    .eq('id', contactId)
  if (error) throw error
}

// Lấy contacts theo nhiều levels (cho campaign)
export async function getContactsByLevels(levelIds) {
  const { data, error } = await supabase
    .from('contacts')
    .select('*, levels(name)')
    .in('level_id', levelIds)
    .eq('status', 'active')
  if (error) throw error
  return data
}
JavaScript — Campaign & Logs
// Tạo campaign mới
export async function createCampaign(payload) {
  const { data, error } = await supabase
    .from('campaigns')
    .insert(payload)
    .select()
    .single()
  if (error) throw error
  return data
}

// Lưu send log sau mỗi email gửi
export async function logSend(campaignId, contact, status, resendId, errorMsg) {
  await supabase.from('send_logs').insert({
    campaign_id: campaignId,
    contact_id:  contact.id,
    email:       contact.email,
    level:       contact.levels?.name,
    status,
    resend_id:   resendId,
    error_msg:   errorMsg
  })
}

// Cập nhật tổng kết campaign khi xong
export async function finalizeCampaign(campaignId, sentCount, failedCount) {
  await supabase
    .from('campaigns')
    .update({
      status: 'completed',
      sent_count:   sentCount,
      failed_count: failedCount,
      sent_at:      new Date().toISOString()
    })
    .eq('id', campaignId)
}

// Lấy send logs của 1 campaign (để export)
export async function getCampaignLogs(campaignId) {
  const { data, error } = await supabase
    .from('send_logs')
    .select('*')
    .eq('campaign_id', campaignId)
    .order('sent_at')
  if (error) throw error
  return data
}

🔐 Security & Row Level Security
RLS đảm bảo chỉ người dùng đã xác thực mới đọc/ghi được data. Đây là lớp bảo mật quan trọng nhất của Supabase.
SQL script RLS đã có ở tab Step 3 → Row Level Security. Chạy script đó sau khi tạo xong tables.
anon key
Dùng ở frontend (public). Bị giới hạn bởi RLS policies. Chỉ đọc được data nếu có policy cho phép.
service_role key
Bypass hoàn toàn RLS. Chỉ dùng ở backend/server. Không bao giờ expose ra frontend!
